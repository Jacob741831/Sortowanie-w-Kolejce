-stworz class sortowanie z structem kolejka *
-konstruktor kolejka robi null kolejka(NULL) 
-destruktor kolejka
-funkcje wczytaj_z_pliku() z liczby.txt, wczytuje ile jest liczb czyli zrób jakąś zmienna która zapisuje ile jest liczb, kazda liczba idzie do wagonika w kolejce
-funckja srotowanie_babelkowe() rosnąco na bazie przypinania wagonika a nie swap() czyli wskaźnik wpinamy czyli na boczku zapisujesz plik i =
-funkcja zapisz_do_pliku() funkcja zapisuje do pliku liczby nazwijmy plik zapisane.txt 
-funkcja wypisz() wypisuje liczby w konsoli 
-zrob menu switch case na bazie int wybor;

2.sortowanie bąbelkowe

Wysyłam pod spodem szkic programu jak coś to popraw typy danych funkcji zależne od tego jakie dane zwracają funkcje też ogólnie popraw cały kod pod te myślniki bo classa też nie jest dobrze napisana 

#include <iostream>
#include <fstream>

using namespace std;

struct kolejka
{
    int a;
    kolejka* next;
};

class sortowanie
{
private:
    kolejka* poczatek;

public:

    // Konstruktor
    sortowanie()
    {
        poczatek = NULL;
    }

    // Destruktor
    ~sortowanie()
    {
        kolejka* temp;

        while (poczatek != NULL)
        {
            temp = poczatek;
            poczatek = poczatek->next;
            delete temp;
        }
    }

    // Wczytywanie liczb z pliku
    void wczytaj_z_pliku()
    {
        ifstream plik("liczby.txt");

        if (!plik)
        {
            cout << "Nie znaleziono pliku liczby.txt!" << endl;
            cout << "Sprawdz, czy plik znajduje sie w odpowiednim folderze." << endl;
            return;
        }

        int ile;
        plik >> ile;

        kolejka* ostatni = NULL;

        for (int i = 0; i < ile; i++)
        {
            int liczba;

            if (!(plik >> liczba))
            {
                cout << "Blad odczytu liczby z pliku!" << endl;
                break;
            }

            kolejka* nowy = new kolejka;

            nowy->a = liczba;
            nowy->next = NULL;

            if (poczatek == NULL)
            {
                poczatek = nowy;
            }
            else
            {
                ostatni->next = nowy;
            }

            ostatni = nowy;
        }

        plik.close();

        cout << "Wczytano " << ile << " liczb." << endl;
    }

    // Sortowanie babelkowe przez przepinanie wezlow
    void sortowanie_babelkowe()
    {
        if (poczatek == NULL || poczatek->next == NULL)
        {
            cout << "Za malo elementow do sortowania." << endl;
            return;
        }

        bool zamiana;

        do
        {
            zamiana = false;

            kolejka* poprzedni = NULL;
            kolejka* aktualny = poczatek;

            while (aktualny != NULL && aktualny->next != NULL)
            {
                kolejka* nastepny = aktualny->next;

                if (aktualny->a > nastepny->a)
                {
                    // Przepinamy wskazniki zamiast uzywac swap()

                    aktualny->next = nastepny->next;
                    nastepny->next = aktualny;

                    if (poprzedni == NULL)
                    {
                        poczatek = nastepny;
                    }
                    else
                    {
                        poprzedni->next = nastepny;
                    }

                    poprzedni = nastepny;

                    zamiana = true;
                }
                else
                {
                    poprzedni = aktualny;
                    aktualny = aktualny->next;
                }
            }

        } while (zamiana);

        cout << "Sortowanie zakonczone." << endl;
    }

    // Zapis do pliku
    void zapisz_do_pliku()
    {
        ofstream plik("zapisane.txt");

        if (!plik)
        {
            cout << "Nie mozna utworzyc pliku zapisane.txt!" << endl;
            return;
        }

        kolejka* aktualny = poczatek;

        while (aktualny != NULL)
        {
            plik << aktualny->a << endl;
            aktualny = aktualny->next;
        }

        plik.close();

        cout << "Zapisano liczby do pliku zapisane.txt." << endl;
    }

    // Wypisywanie liczb
    void wypisz()
    {
        if (poczatek == NULL)
        {
            cout << "Kolejka jest pusta." << endl;
            return;
        }

        kolejka* aktualny = poczatek;

        cout << "Liczby: ";

        while (aktualny != NULL)
        {
            cout << aktualny->a << " ";
            aktualny = aktualny->next;
        }

        cout << endl;
    }
};


int main()
{
    sortowanie sort;

    cout << "Program SORTOWANIE W KOLEJCE" << endl;
    cout << "============================" << endl;

    sort.wczytaj_z_pliku();

    int wybor;

    do
    {
        cout << endl;
        cout << "===== MENU =====" << endl;
        cout << "1. Wypisz liczby" << endl;
        cout << "2. Sortowanie babelkowe" << endl;
        cout << "3. Zapisz do pliku" << endl;
        cout << "0. Wyjscie" << endl;
        cout << "================" << endl;

        cout << "Wybierz: ";
        cin >> wybor;

        switch (wybor)
        {
            case 1:
                sort.wypisz();
                break;

            case 2:
                sort.sortowanie_babelkowe();
                break;

            case 3:
                sort.zapisz_do_pliku();
                break;

            case 0:
                cout << "Koniec programu." << endl;
                break;

            default:
                cout << "Nieprawidlowy wybor!" << endl;
        }

    } while (wybor != 0);

    return 0;
}
